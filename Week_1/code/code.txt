#include "DHTesp.h"

// ===== 腳位設定 =====
// 如果改了腳位，這裡也要同步修改腳位設定
const int DHT_PIN = 23;       // DHT22 資料腳
const int SOIL_PIN = 34;      // 可變電阻 SIG，模擬土壤濕度
const int LIGHT_PIN = 35;     // LDR 的 AO，讀取光照
const int LED_PIN = 25;       // 警告 LED
const int BUZZER_PIN = 26;    // 蜂鳴器正極

// ===== 可以自行調整的警告設定 =====
const int SOIL_ALERT_PERCENT = 20;  // 土壤濕度低於幾 % 要警告
const int BLINK_COUNT = 4;          // 每輪警告讓 LED 閃爍幾次
const int BUZZER_FREQ = 1000;       // 蜂鳴器音高，數字越大聲音越尖（Hz）
const int ALERT_ON_MS = 300;        // LED 亮、蜂鳴器響多久（毫秒）
const int ALERT_OFF_MS = 300;       // LED 暗、蜂鳴器停止多久（毫秒）
const int NORMAL_WAIT_MS = 2000;    // 濕度正常時，每隔多久重新讀取（毫秒）

DHTesp dht; // 建立一個 DHT22 感測器物件

void setup() {
  // 開啟 Serial Monitor；監控視窗的速度也要設為 115200
  Serial.begin(115200);

  // 使用 12 位元類比讀值，範圍為 0～4095
  analogReadResolution(12);

  // 告訴程式 DHT22 接在哪個腳位
  dht.setup(DHT_PIN, DHTesp::DHT22);

  // LED 與蜂鳴器都是由 ESP32 控制的輸出
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  // 程式啟動時先關閉警告
  digitalWrite(LED_PIN, LOW);
  noTone(BUZZER_PIN);
}

void loop() {
  // 一次讀取 DHT22 的溫度與空氣濕度
  TempAndHumidity data = dht.getTempAndHumidity();

  // 讀取可變電阻和 LDR 的原始數值
  int soilRaw = analogRead(SOIL_PIN);
  int lightRaw = analogRead(LIGHT_PIN);

  // 將 0～4095 的原始數值換算成 0～100%
  // 這裡約定：旋鈕數值越大，模擬的土壤濕度越高
  int soilPercent = map(soilRaw, 0, 4095, 0, 100);

  // 此 LDR 模組光線越強，AO 數值通常越低，所以反向換算
  // 這是方便觀察的相對百分比，不是真正的 lux
  int lightPercent = map(lightRaw, 0, 4095, 100, 0);

  Serial.println("------ 感測器數值 ------");

  // 若 DHT22 讀取失敗，避免顯示無效數值
  if (isnan(data.temperature) || isnan(data.humidity)) {
    Serial.println("DHT22：讀取失敗");
  } else {
    Serial.print("溫度：");
    Serial.print(data.temperature, 1); // 1 表示顯示到小數點後 1 位
    Serial.println(" °C");

    Serial.print("空氣濕度：");
    Serial.print(data.humidity, 1);
    Serial.println(" %");
  }

  Serial.print("土壤濕度（模擬）：");
  Serial.print(soilPercent);
  Serial.println(" %");

  Serial.print("光照強度（相對值）：");
  Serial.print(lightPercent);
  Serial.println(" %");

  // 只有「低於」設定值才警告；剛好等於 20% 不會警告
  if (soilPercent < SOIL_ALERT_PERCENT) {
    Serial.println("警告：土壤濕度過低！");

    // 重複執行「亮燈並發聲 → 關燈並停止發聲」
    for (int i = 0; i < BLINK_COUNT; i++) {
      digitalWrite(LED_PIN, HIGH);
      tone(BUZZER_PIN, BUZZER_FREQ);
      delay(ALERT_ON_MS);

      digitalWrite(LED_PIN, LOW);
      noTone(BUZZER_PIN);
      delay(ALERT_OFF_MS);
    }
  } else {
    // 土壤濕度正常時，確保 LED 和蜂鳴器都關閉
    digitalWrite(LED_PIN, LOW);
    noTone(BUZZER_PIN);
    Serial.println("土壤濕度正常");

    delay(NORMAL_WAIT_MS);
  }

  Serial.println(); // 印一個空白行，讓每輪數值比較容易閱讀
}
